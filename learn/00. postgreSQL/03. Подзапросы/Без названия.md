Все уникальные компании заказчиков, которые делали заказы на более чем 40 единиц товаров

SELECT DISTINCT c.company_name
FROM customers AS c
JOIN orders AS o USING(customer_id)
WHERE EXISTS (SELECT order_id FROM order_details
				WHERE order_id = o.order_id
				AND quantity > 40)


SELECT DISTINCT c.company_name
FROM customers AS c
JOIN orders AS o USING(customer_id)
JOIN order_details AS od USING(order_id)
WHERE od.quantity > 40


SELECT DISTINCT c.company_name
FROM customers AS c
JOIN orders AS o USING(customer_id)

SELECT DISTINCT c.company_name
FROM customers AS c
WHERE customer_id = ANY(
		SELECT customer_id
		FROM order
		JOIN order_details USING(order_id)
		WHERE od.quantity > 40
)
